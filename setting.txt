# 1. 가상환경 생성을 위한 패키지 설치
sudo apt update
sudo apt install python3-venv python3-full

# 2. 프로젝트 폴더로 이동 (이미 있다면 생략)
# cd ~/your_project_folder

# 3. 'venv'라는 이름의 가상환경 생성
python3 -m venv venv

# 4. 가상환경 활성화
source venv/bin/activate

# 가상환경 내 pip 업데이트
pip install --upgrade pip
